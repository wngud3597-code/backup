- 비전공자도 이해할 수 있는 말로


# 점심 메뉴 투표 집계기
ADMIN_PW = "admin1234"

def count(votes):
    r = {}
    for v in votes:
        if v in r:
            r[v] += 1
        else:
            r[v] = 1
    return r

def winner(votes):
    r = count(votes)
    return max(r, key=r.get)

def rate(votes, menu):
    return count(votes)[menu] / len(votes) * 100

def reset(pw):
    if pw == ADMIN_PW:
        print("초기화 완료")

votes = input("투표(쉼표로 구분): ").split(",")
print("1등 메뉴:", winner(votes))
print("국밥 득표율:", rate(votes, "국밥"), "%")
